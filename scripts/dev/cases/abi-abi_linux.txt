# C4: the Linux registry (kernel/abi/abi_linux.c), proven by breaking it.
#
# The properties are held by test_abi_vocabulary_and_linux in the host suite: a
# number claimed twice - inside one table or across two - is refused and leaves the
# registry untouched; a row with no handler is refused; and a declared operation
# with no row is named by vibeos_abi_linux_missing. The rows of the *real* tables
# are held by check-syscall-checks.py, whose sabotage cases are in core-syscall.txt.
#
# Target source: kernel/abi/abi_linux.c
# Run: python3 scripts/dev/sabotage.py kernel/abi/abi_linux.c scripts/dev/cases/abi-abi_linux.txt build-clang-Release

        if (find_row(rows[i].nr)) {

        for (j = 0; j < i; j++) {

        }
+++ new
