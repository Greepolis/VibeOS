# Permission checks in the file handlers (docs/abi/ L2 step 1). Host test:
# t_credentials in tests/kernel/linux_abi_tests.c.
#
#  1. an open does not ask whether the file may be read or written
#  2. a new name is made in a directory the caller may not write
#  3. a name is removed from one
#  4. the sticky bit is ignored
#  5. a new file is root's whoever made it
#  6. truncate does not ask
#  7. chdir into a directory the caller may not search
#  8. the walk does not ask for search permission
#
# Target source: kernel/abi/linux/fs.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/abi/linux/fs.c scripts/dev/cases/abi-cred.txt

### an open does not ask whether the file may be read or written
--- old
        err = linux_may(&w.node, want);
    } else {
+++ new

        (void)want;
    } else {

### a new name is made in a directory the caller may not write
--- old

}
+++ new
    return 0;

+++ new

### the sticky bit is ignored
--- old
