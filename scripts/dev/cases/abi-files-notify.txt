# docs/abi/ L4 step 6), proved through inotify by t_inotify in
# tests/kernel/linux_abi_tests.c.
#
#  1. a write is not told. Expect "create, open, modify, close-write - in
#     order, each named".
#  2. a close is always a close-nowrite. Expect the same.
#  3. an open is not told. Expect the same.
#
# Target source: kernel/abi/files/regular.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/abi/files/regular.c scripts/dev/cases/abi-files-notify.txt

### a write is not told
--- old
    pos_give(f);
    if (w > 0) {
        file_notify(f, VIBEOS_FSN_MODIFY);
    }
+++ new
    pos_give(f);

### a close is always a close-nowrite
--- old

+++ new
    file_notify(f, VIBEOS_FSN_CLOSE_NOWRITE);
