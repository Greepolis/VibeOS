# The file types every personality shares (kernel/abi/files/, docs/abi/ A3), proved
# by the Linux handlers' host tests (tests/kernel/linux_abi_tests.c).
#
#  1. a pipe ignores O_NONBLOCK and waits. Expect "an empty non-blocking pipe
#     answers EAGAIN instead of waiting".
#  2. a pipe says it is a regular file - what fstat answered before A3, when the
#     table entry was all there was to ask. Expect "fstat says FIFO".
#  3. a regular file's read keeps its own position instead of the description's:
#     the offset a dup shares stops moving. Expect "the duplicate continues where
#     the original stopped".
#
# Target source: kernel/abi/files/pipefile.c (cases 1 and 2); kernel/abi/files/regular.c for case 3
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/abi/files/pipefile.c scripts/dev/cases/abi-files.txt

### a pipe ignores O_NONBLOCK
--- old
        if (f->flags & VIBEOS_O_NONBLOCK) {
            return -VIBEOS_EAGAIN;
        }
+++ new

### a pipe says it is a regular file
--- old
    out->mode = VIBEOS_S_IFIFO | 0600u;
+++ new
    out->mode = VIBEOS_S_IFREG | 0644u;

### a regular file's read does not advance the shared position
--- old
    long n = regular_pread(f, buf, len, f->pos);
    if (n > 0) {
        f->pos += (uint64_t)n;
    }
+++ new
    long n = regular_pread(f, buf, len, f->pos);
