### fork copies the break after releasing the address-space lock
--- old
    cps->brk_cur = pps->brk_cur;
    __atomic_store_n(&cps->mmap_cur,
                     __atomic_load_n(&pps->mmap_cur, __ATOMIC_ACQUIRE),
                     __ATOMIC_RELEASE);
    /* The parent's tables and list have both been read; release before the
     * child-only setup below. */
    ks_mm_unlock(pps);
+++ new
    __atomic_store_n(&cps->mmap_cur,
                     __atomic_load_n(&pps->mmap_cur, __ATOMIC_ACQUIRE),
                     __ATOMIC_RELEASE);
    ks_mm_unlock(pps);
    cps->brk_cur = pps->brk_cur;

### fork copies the mapping cursor after releasing it
--- old
    cps->brk_cur = pps->brk_cur;
    __atomic_store_n(&cps->mmap_cur,
                     __atomic_load_n(&pps->mmap_cur, __ATOMIC_ACQUIRE),
                     __ATOMIC_RELEASE);
    /* The parent's tables and list have both been read; release before the
     * child-only setup below. */
