#  1. a thread's directory describes its process's leader: task/<tid>/stat

#     main thread waits shows R while another thread reads - futex_wait03

### a process's state is any running thread's

+++ new

    running |= me >= 0 && ks_id(me)->tgid == t->tgid && !ks_id(me)->sleeping;
