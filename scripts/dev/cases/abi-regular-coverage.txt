### O_APPEND writes at the current end, not the descriptor offset
--- old
    at = (f->flags & VIBEOS_O_APPEND) ? regular_size(f) : f->pos;
+++ new
    at = f->pos;

### ftruncate changes the file's visible size
--- old
    r = vibeos_fs_truncate(f->mnt, &node, size);
    if (r == 0) {
        f->size = size;
+++ new
    r = 0;
    if (r == 0) {
        f->size = size;
