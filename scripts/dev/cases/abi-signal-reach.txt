# Who may signal whom, and through what (docs/abi/ L2 step 7). Host tests:
# t_kill_permission and t_procdev in tests/kernel/linux_abi_tests.c.
#
#  1. a session is a wall again: the superuser's kill(pid, 0) from another
#     session is refused - BusyBox's timeout then believes its program ended
#  2. SIGCONT within a session is judged by owners like any other signal
#  3. a group whose members nobody may signal answers ESRCH, as if empty
#  4. an open /proc/<pid> is not taken for a pidfd
#
# Target source: kernel/abi/linux/sig.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/abi/linux/sig.c scripts/dev/cases/abi-signal-reach.txt

### a session is a wall again
--- old
    if (sig == VIBEOS_SIGCONT && ks_id(target)->sid == me->sid) {
        return 1;
    }
+++ new
    if (ks_id(target)->sid != me->sid) {
        return 0;
    }
    if (sig == VIBEOS_SIGCONT && ks_id(target)->sid == me->sid) {
        return 1;
    }

### SIGCONT within a session is judged by owners
--- old
    if (sig == VIBEOS_SIGCONT && ks_id(target)->sid == me->sid) {
+++ new
    if (sig == 0xFFFFu && ks_id(target)->sid == me->sid) {

### a group nobody may be sent to answers ESRCH
--- old
        return delivered != 0 ? 0 : members != 0 ? -VIBEOS_EPERM : -VIBEOS_ESRCH;
+++ new
        return delivered != 0 ? 0 : -VIBEOS_ESRCH;

### an open /proc/<pid> is not a pidfd
--- old
    pid = f->ops == &vibeos_fops_pidfd ? f->proc_pid : vibeos_procfs_node_pid(f->mnt, f->node);
+++ new
    pid = f->ops == &vibeos_fops_pidfd ? f->proc_pid : 0u;
