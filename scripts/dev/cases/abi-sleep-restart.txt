# Sleeping, and a call run again after a signal (docs/abi/ L3 step 3). Host
# tests: t_sleep and t_sa_restart in tests/kernel/linux_abi_tests.c.
#
#  1. a sleep does not count the tick it began in (it can end early)
#  2. a time shorter than a tick is no wait at all
#  3. a signal does not end a sleep
#  4. what was left of an interrupted sleep is not reported
#  5. every clock is slept on as the wall clock: a sleep on the process's CPU
#     time ends when the wall clock has moved that far (L2 step 7)
#  6. the clocks Linux cannot sleep on answer EINVAL, not EOPNOTSUPP
#
# Target source: kernel/abi/linux/misc.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/abi/linux/misc.c scripts/dev/cases/abi-sleep-restart.txt

### a sleep does not count the tick it began in
--- old
        deadline = (uint64_t)linux_clock_read(clk) + (uint64_t)ticks + 1u;
+++ new
        deadline = (uint64_t)linux_clock_read(clk) + (uint64_t)ticks;

### a time shorter than a tick is no wait at all
--- old
    return (int64_t)((uint64_t)ts->tv_sec * ks_hz() + ((uint64_t)ts->tv_nsec + per - 1u) / per);
+++ new
    return (int64_t)((uint64_t)ts->tv_sec * ks_hz() + (uint64_t)ts->tv_nsec / per);
