# The options every socket keeps (kernel/abi/files/sockopt.c, docs/abi/ L5
# step 1), proved by t_sock_bsd in tests/kernel/linux_abi_tests.c.
#
#  1. a buffer size is not doubled. Expect "SO_SNDBUF is doubled".
#  2. a buffer size has no floor. Expect "SO_RCVBUF is never below its floor".
#     The first check set 1000, whose double is under the floor too, so case 1
#     could not have gone red against it; the SO_SNDBUF check is why.
#
# Target source: kernel/abi/files/sockopt.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/abi/files/sockopt.c scripts/dev/cases/abi-sockopt.txt

### a buffer size is not doubled
--- old
            u = u * 2u < floor ? floor : u * 2u;
+++ new
            u = u < floor ? floor : u;

### a buffer size has no floor
--- old
            u = u * 2u < floor ? floor : u * 2u;
+++ new
            u = u * 2u;
