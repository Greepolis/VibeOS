# What a pipe can do without waiting (kernel/abi/files/pipefile.c, docs/abi/ L1
# step 7), proved through poll by tests/kernel/linux_abi_tests.c.
#
#  1. a pipe nobody can write says nothing. Expect "an empty pipe nobody can
#     write any more is hung up, and only that" (L4 step 1: it used to be
#     readable as well, which Linux does not say - LTP's poll03).
#  2. an empty pipe is readable. Expect "an empty pipe: nothing to read, room
#     to write".
#  3. a pipe nobody reads is a hangup to its writer, not an error (docs/abi/
#     L4). Expect "a pipe nobody reads any more is writable - the write fails at
#     once - and an error, not a hangup".
#
# Target source: kernel/abi/files/pipefile.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/abi/files/pipefile.c scripts/dev/cases/abi-tty-pipe.txt

### a pipe nobody can write says nothing
--- old
    return vibeos_pipe_writers(f->pipe) == 0u ? VIBEOS_READY_HUP : 0u;
+++ new
    return 0u;

### an empty pipe is readable
--- old
    if (pending > 0u) {
        return VIBEOS_READY_IN;
    }
+++ new
    if (1) {
        return VIBEOS_READY_IN;
    }

### a pipe nobody reads is a hangup, not an error
--- old
        return vibeos_pipe_readers(f->pipe) == 0u ? (r | VIBEOS_READY_OUT | VIBEOS_READY_ERR) : r;
+++ new
        return vibeos_pipe_readers(f->pipe) == 0u ? (r | VIBEOS_READY_OUT | VIBEOS_READY_HUP) : r;
