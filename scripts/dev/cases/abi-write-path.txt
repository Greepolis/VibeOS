# Writing through a descriptor (kernel/abi/files/regular.c, docs/abi/ L1 step 3),
# proved by the handlers' host tests in tests/kernel/linux_abi_tests.c on the
# fake's tmpfs at /tmp.
#
#  1. O_EXCL ignored. Expect "O_EXCL refuses an existing file".
#  (7.) pwrite under O_APPEND writes at its offset, as POSIX says and Linux
#     does not. Expect "under O_APPEND pwrite64 writes at the end" (L4 step 1,
#     LTP's pwrite04 - one of L1's twelve its list never had).
#  2. O_TRUNC ignored. Expect "O_TRUNC empties the file at open".
#  3. O_APPEND writes at the description's position. Expect "O_APPEND writes at
#     the end" or "two appenders interleave"; at boot, tmp_write_path_failed -
#     the shell's >> overwrites the start of the file and wc counts 820.
#  4. pread reads at the position, not the offset. Expect "pread64 reads
#     there".
#  5. a write is not counted as progress past a short one - the loop goes on
#     after the filesystem took less than it was given. Left out: nothing here
#     makes tmpfs take less without failing, so no test could tell.
#  6. ftruncate allowed on a read-only descriptor. Expect "ftruncate on a
#     read-only descriptor is EINVAL". NOT RED when first run, and for a reason
#     that was a defect: a read-only description was not marked as writing in
#     place, so it fell into the whole-file branch and was refused there
#     instead - and its fstat and SEEK_END reported the size from when it was
#     opened. (That branch is gone since step 4: every regular file asks the
#     filesystem.)
#  7. a new file ignores the mode it was given. Expect "less the umask".
#
# Target source: kernel/abi/files/regular.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/abi/files/regular.c scripts/dev/cases/abi-write-path.txt

### O_EXCL ignored
--- old
        if ((flags & VIBEOS_O_CREAT) && (flags & VIBEOS_O_EXCL)) {
            *err = -VIBEOS_EEXIST;
            return 0;
        }
+++ new

### O_TRUNC ignored
--- old
            if (r == 0 && (flags & VIBEOS_O_TRUNC)) {
+++ new

### O_APPEND writes at the description's position
--- old
    at = (f->flags & VIBEOS_O_APPEND) ? regular_size(f) : f->pos;
+++ new
    at = f->pos;

### pread reads at the position, not the offset
--- old
        got = vibeos_fs_read_at(f->mnt, &node, off + done, bounce, (uint32_t)want);
+++ new
        got = vibeos_fs_read_at(f->mnt, &node, f->pos + done, bounce, (uint32_t)want);

### ftruncate allowed on a read-only descriptor
--- old
        return -VIBEOS_EINVAL;   /* Linux: not open for writing */
+++ new
        (void)0;

### a new file ignores the mode it was given
--- old
        r = vibeos_fs_create(mnt, tail, mode & 07777u, &node);
+++ new
        r = vibeos_fs_create(mnt, tail, 0644u, &node);

    return regular_pwrite_direct(f, buf, len, off);
+++ new
