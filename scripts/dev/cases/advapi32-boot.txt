### RegCreateKeyEx makes only the last key of a path
--- old
        st = NtCreateKey(&next, last ? (sam & ~KEY_WOW64_MASK) : MAXIMUM_ALLOWED, oa, 0, 0, options & 1u,
                         &d);
+++ new
