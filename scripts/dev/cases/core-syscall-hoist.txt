# C4 stage 2b: pointer arguments declared in the row, validated before the handler.
#
# uname, time and clock_gettime keep no check of their own; the row's PTRS(...) is
# all there is. So removing the descriptor leaves a handler that writes a user
# pointer nobody validated - check-syscall-checks.py must see USER_MEMORY vanish from
# the operation. (core-ptr-descriptors.txt does the same for the rest of the rows,
# and core-ptr-engine.txt breaks the engine itself.)
#
# Target source: kernel/abi/linux/misc.c
# Run: SABOTAGE_VERIFY="python3 scripts/dev/check-syscall-checks.py" python3 scripts/dev/sabotage.py kernel/abi/linux/misc.c scripts/dev/cases/core-syscall-hoist.txt

### uname's row stops declaring its buffer
--- old
PTRS(OUT(0, 6u * 65u)), hw_sys_uname(ARG(0))
+++ new
NOPTR, hw_sys_uname(ARG(0))

### clock_gettime's row stops declaring its timespec
--- old
PTRS(OUT(1, 16)), hw_sys_clock_gettime(ARG(0), ARG(1))
+++ new
NOPTR, hw_sys_clock_gettime(ARG(0), ARG(1))

### time's row stops declaring its optional pointer
--- old
PTRS(OUT_OPT(0, 8)), hw_sys_time(ARG(0))
+++ new
NOPTR, hw_sys_time(ARG(0))
