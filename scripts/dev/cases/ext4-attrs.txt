# the torture compares each with the host's tree - mode and owner on every
# file and directory, mtime where utimes set it - so each must go red.
#
# Target source: kernel/fs/ext4_write.c
# Run: SABOTAGE_VERIFY="bash -c 'cmake --build build-gcc-Release --target vibeos_ext4_exercise > /dev/null && bash scripts/dev/verify-ext4-torture.sh build-gcc-Release'" python3 scripts/dev/sabotage.py kernel/fs/ext4_write.c scripts/dev/cases/ext4-attrs.txt build-gcc-Release

### chown keeps only the low half of the owner
--- old
        st16(inode + 0x78u, a->uid >> 16);
+++ new
        st16(inode + 0x78u, 0);

### chown sets the group to the owner
--- old
        st16(inode + 0x18u, a->gid);
+++ new
        st16(inode + 0x18u, a->uid);

### utimes leaves the mtime
--- old
        time_set(inode, T_MTIME, a->mtime_ns);
+++ new
        (void)0;

### chmod drops the group's bits
--- old
        st16(inode, (le16(inode) & VIBEOS_S_IFMT) | (a->mode & 07777u));
+++ new
        st16(inode, (le16(inode) & VIBEOS_S_IFMT) | (a->mode & 07707u));
