# What the file layer says happened (kernel/fs/vfs.c, docs/abi/ L4 step 6),
# proved through inotify by t_inotify in tests/kernel/linux_abi_tests.c.
#
#  1. a rename's halves get different cookies. Expect "MOVED_FROM and
#     MOVED_TO to the directory with one cookie; ...".
#  2. an unlink does not say the link count changed. Expect "unlink: ATTRIB to
#     the file, DELETE to the directory, DELETE_SELF and IGNORED".
#  3. a mkdir is not told. Expect "CREATE with IN_ISDIR".
#  4. chmod is not told. Expect "the file's own watch hears ATTRIB without a
#     name; the directory did not ask for it".
#
# Target source: kernel/fs/vfs.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/fs/vfs.c scripts/dev/cases/fs-notify.txt

### a rename's halves get different cookies
--- old
        vibeos_fs_notify(mnt, to, n.id, VIBEOS_FSN_MOVED_TO, cookie, n.is_dir);
+++ new
        vibeos_fs_notify(mnt, to, n.id, VIBEOS_FSN_MOVED_TO, cookie + 1u, n.is_dir);

### an unlink does not say the link count changed
--- old
    if (!n->is_dir) {
        vibeos_fs_notify(mnt, path, n->id, VIBEOS_FSN_NLINK, 0, 0);
    }
+++ new

### a mkdir is not told
--- old
    r = mnt->ops->mkdir(mnt->fs, path);
    if (r == 0) {
        fs_notify_made(mnt, path);
    }
+++ new
    r = mnt->ops->mkdir(mnt->fs, path);

### chmod is not told
--- old
    if (r == 0 && fs_peek(mnt, path, &n)) {
        vibeos_fs_notify(mnt, path, n.id, VIBEOS_FSN_ATTRIB, 0, n.is_dir);
    }
+++ new
