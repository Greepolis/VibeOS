# /proc (docs/abi/ L3 step 3). Host tests: t_procfs in
# tests/kernel/linux_abi_tests.c.
#
#  1. a number is printed without its unit and newline, so a reader's scanf
#     matches the next line's name as part of this one's value
#  2. a file's size is not what a read of it returns
#  3. half a directory's name is taken for the directory
#
# Not a case today: "a directory lists a name once for every file under it".
# No directory here holds two files yet, so listing without the check prints
# the same entries; the first second file under /proc/sys/kernel makes it one.
#
# Target source: kernel/fs/procfs.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/fs/procfs.c scripts/dev/cases/fs-procfs.txt

### a number is printed without its unit and newline
--- old
    pf_char(o, ' ');
    pf_char(o, 'k');
    pf_char(o, 'B');
    pf_char(o, '\n');
+++ new
    pf_char(o, ' ');

### a file's size is not what a read of it returns
--- old
            out->size = g_files[i].text((const vibeos_procfs_t *)fs, text, PF_TEXT_MAX);
            out->is_dir = 0;
+++ new
            out->size = 4096u;
            (void)text;
            out->is_dir = 0;

### half a directory's name is taken for the directory
--- old
    return (path[i] == 0 && f[i] == '/') ? (int)(i + 1u) : -1;
+++ new
    return (path[i] == 0) ? (int)(i + 1u) : -1;
