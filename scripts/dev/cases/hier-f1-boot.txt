# The hierarchy's F1 on the machine (docs/filesystems/hierarchy/inventory.md):
# /run and /dev/shm are tmpfs. The self-test writes a file to each and reads
# it back; the gate reads both answers off the program's output lines, and
# afterwards asks debugfs whether /run left the guest's file on the disk.
#
#  1. /run is not mounted: the file lands in the root's directory. Expect
#     run_is_not_volatile.
#  2. /run has no room: the write fails. Expect run_round_trip_failed.
#  3. /dev/shm is not mounted: devfs makes no file. Expect
#     dev_shm_round_trip_failed.
#
# The directory contract's modes are hier-f1-modes.txt.
#
# Target source: kernel/arch/x86_64/io_bringup.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-boot.sh" python3 scripts/dev/sabotage.py kernel/arch/x86_64/io_bringup.c scripts/dev/cases/hier-f1-boot.txt

### /run has no room (run_round_trip_failed)
--- old
        uint64_t pages = e->size_kib ? (e->size_kib + 3u) / 4u : HW_TMPFS_PAGES;
+++ new
        uint64_t pages = e->size_kib ? 1u : HW_TMPFS_PAGES;
