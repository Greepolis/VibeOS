# The console's portable halves (vibeos/vt.h): output sequences and scan codes.
# Read by test_vt in the host tests.
#
# S7 (docs/shell/s7.md) added what a full-screen program needs, and four cases
# for it, each expecting a test in test_output:
#   - an OSC is text: "a title, ended by BEL, prints nothing";
#   - UTF-8 is read as bytes: "a three-byte character is one code point";
#   - an escape inside a UTF-8 character is lost: "a character cut short by an
#     escape keeps the escape";
#   - an SGR keeps two numbers: "an SGR keeps every number".
#
# Target source: kernel/io/vt.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/io/vt.c scripts/dev/cases/io-vt.txt

### a position is taken as counted from zero
--- old
            *n = a ? a - 1u : 0u;
            *m = b ? b - 1u : 0u;
+++ new
            *n = a;
            *m = b ? b - 1u : 0u;

### a large number wraps instead of saturating
--- old
            vt->param[vt->count] = v > 0xFFFFu ? 0xFFFFu : (uint16_t)v;
+++ new
            vt->param[vt->count] = (uint16_t)v;
