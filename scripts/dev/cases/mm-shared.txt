# Shared pages in the address-space layer (docs/abi/ L3 step 2). Host tests:
# t_mmap_shared in tests/kernel/linux_abi_tests.c, which runs the handlers over
# this layer.
#
#  1. a mapping asked to be shared is not marked
#  2. fork makes a shared page copy-on-write like any other
#  3. fork gives the child the frame without the mark or the write bit
#  4. a shared page is paged out like any other
#
# Target source: kernel/mm/vmspace.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host.sh build-gcc-Release" python3 scripts/dev/sabotage.py kernel/mm/vmspace.c scripts/dev/cases/mm-shared.txt

### a mapping asked to be shared is not marked
--- old
    if (prot & VIBEOS_PROT_SHARED) {
        f |= VIBEOS_PTE_SHARED;
    }
+++ new

### fork makes a shared page copy-on-write like any other
--- old
        if (entry & VIBEOS_PTE_SHARED) {
            break;

+++ new

### fork gives the child the frame without the mark or the write bit
--- old

### a shared page is paged out like any other
