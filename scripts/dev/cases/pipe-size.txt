# A pipe's own size (docs/abi/ L7): a ring of pages, 64 KiB by default,
# resized by F_SETPIPE_SZ. Host tests: test_pipe in tests/kernel/pipe_tests.c.
#
#  1. a resize leaves the bytes where they were in the old ring. Expect "what
#     it held comes out in order after the resize".
#  2. a resize smaller than what is held is taken. Expect "a size smaller than
#     what it holds is refused".
#  3. a size is not rounded to a power of two pages. Expect "a size is rounded
#     up to a power of two pages".
#
# Target source: kernel/ipc/pipe.c
# Run: SABOTAGE_VERIFY="bash scripts/dev/verify-host-quick.sh" python3 scripts/dev/sabotage.py kernel/ipc/pipe.c scripts/dev/cases/pipe-size.txt

### a resize leaves the bytes where they were in the old ring
--- old
        p->head = (p->head + run) % p->cap;
        left -= run;
+++ new
        left -= run;

### a resize smaller than what is held is taken
--- old
    if (!p || p->count > pages * PIPE_PAGE) {
        unlock();
+++ new
    if (!p) {
        unlock();

### a size is not rounded to a power of two pages
--- old
        pages *= 2u;   /* a power of two pages, as Linux's round_pipe_size */
+++ new
        pages += 1u;
