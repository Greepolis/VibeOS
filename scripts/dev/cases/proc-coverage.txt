# Target source: kernel/proc/process.c

### reap leaves the terminated process slot allocated
--- old
    entry->pid = 0;
    entry->parent_pid = 0;
    entry->process_group_id = 0;
    entry->session_id = 0;
    entry->session_leader = 0;
    entry->in_use = 0;
    entry->state = VIBEOS_PROCESS_STATE_TERMINATED;
+++ new
    entry->pid = 0;
    entry->parent_pid = 0;
    entry->process_group_id = 0;
    entry->session_id = 0;
    entry->session_leader = 0;
    entry->in_use = 1;
    entry->state = VIBEOS_PROCESS_STATE_TERMINATED;

### terminate fails to reparent children to init
--- old
            pt->entries[i].parent_pid = (pid == 1u) ? 0u : 1u;
+++ new
            pt->entries[i].parent_pid = 0u;

### terminate retains an owned thread slot
--- old
            pt->threads[i].in_use = 0;
+++ new
            pt->threads[i].in_use = 1;
