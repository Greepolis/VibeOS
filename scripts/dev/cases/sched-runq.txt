### the queue picks a slot without asking whether it may run
--- old
        if (q->runnable(q->ctx, slot, cpu)) {
            q->cursor[cpu] = (slot + 1u) % q->slots;
            return (int)slot;
        }
+++ new
        q->cursor[cpu] = (slot + 1u) % q->slots;
        return (int)slot;

### the scan restarts from zero, so the high slots starve
--- old
        q->cursor[cpu] = (slot + 1u) % q->slots;
+++ new
        q->cursor[cpu] = 0;

### a task that has stopped being runnable keeps its CPU
--- old

    if (current >= 0) {
        return -1;
    }

### the current task is picked again instead of rotating
--- old
        if ((int)slot == current) {

+++ new
