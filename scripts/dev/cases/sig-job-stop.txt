### wait never reports a stop
--- old
                ((t->stop_report == VIBEOS_STOP_REPORT_STOPPED && (options & LINUX_WSTOPPED)) ||
+++ new
                ((0 && (options & LINUX_WSTOPPED)) ||

### a stop is reported without being asked for
--- old
                ((t->stop_report == VIBEOS_STOP_REPORT_STOPPED && (options & LINUX_WSTOPPED)) ||
+++ new
                ((t->stop_report == VIBEOS_STOP_REPORT_STOPPED) ||

### a stop is reported for ever
--- old
                if (reap) {
                    t->stop_report = VIBEOS_STOP_REPORT_NONE;
                }
+++ new

### the stop's status word is the exit's
--- old
                out->status = out->event == VIBEOS_STOP_REPORT_STOPPED ? (int)(0x7fu | ((uint32_t)t->stop_sig << 8))
+++ new
                out->status = out->event == VIBEOS_STOP_REPORT_STOPPED ? (int)((uint32_t)t->stop_sig << 8)

        si.si_code = w.event == VIBEOS_STOP_REPORT_STOPPED     ? LINUX_CLD_STOPPED
