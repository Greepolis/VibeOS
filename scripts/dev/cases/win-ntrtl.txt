# The Rtl runtime of our ntdll (user/win/ntdll/rtl/, docs/abi/windows.md W2), broken
# one rule at a time. Each is caught by tests/win/ntrtl_tests*.c on the host,
# which compiles the same files ntdll.dll is built from.
#
# One source file per run; sabotage.py reports the cases that belong to the other
# files as "anchor not found", which is expected here and not a defect. The
# sections below are marked with the file they break.
#
# Target source: user/win/ntdll/rtl/heap.c; user/win/ntdll/rtl/string.c; user/win/ntdll/rtl/unicode.c; user/win/ntdll/rtl/misc.c; user/win/ntdll/rtl/ip.c; user/win/ntdll/rtl/handle.c; user/win/ntdll/rtl/image.c; user/win/ntdll/rtl/env.c; user/win/ntdll/rtl/critsec.c; user/win/ntdll/rtl/path.c; user/win/ntdll/rtl/printf.c; user/win/ntdll/rtl/dbgprint.c
# Run: for f in heap string unicode misc ip handle image env critsec path printf dbgprint; do SABOTAGE_VERIFY="bash scripts/dev/verify-ntrtl.sh build-gcc-Release" python3 scripts/dev/sabotage.py user/win/ntdll/rtl/$f.c scripts/dev/cases/win-ntrtl.txt build-gcc-Release; done
